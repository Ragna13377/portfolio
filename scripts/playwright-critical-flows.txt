async (page) => {
  const check = (condition, message) => { if (!condition) throw new Error(message); };
  const issues = [];
  page.on('pageerror', error => issues.push(error.message));
  await page.setViewportSize({ width: 1600, height: 900 });
  await page.evaluate(() => localStorage.setItem('portfolio-locale', 'en'));
  await page.reload();
  const crt = page.locator('[data-hardware="crt"]');
  const rom = page.locator('[data-theme]');
  const choices = () => crt.locator('nav button');
  const power = page.locator('[data-console-control="power"]');
  const reset = page.locator('[data-console-control="reset"]');
  const wait = state => page.locator(`[data-power-state="${state}"]`).waitFor();
  const main = () => page.locator('[data-screen="main"]').waitFor();
  const back = () => crt.locator('[data-rom-back]').click();
  const animation = () => crt.locator('[aria-label="CRT viewport"] > div').first().evaluate(el => getComputedStyle(el).animationName);
  const lineAnimation = () => crt.locator('[aria-label="CRT viewport"] > div').last().evaluate(el => getComputedStyle(el, '::after').animationName);
  await wait('on');
  check(await rom.getAttribute('data-world-label') === null, 'World captions removed');
  check(await crt.locator('main, [aria-hidden="true"]').allTextContents().then(texts => texts.every(text => !text.includes('✦ ✦ ✦') && !text.includes(' / HOME') && !text.includes(' / ARCHIVE'))), 'Decorative CRT HUD removed');
  const heroOrder = await crt.locator('[data-frame]').evaluateAll(frames => {
    const animations = frames.map(frame => frame.getAnimations()[0]);
    animations.forEach(animation => animation.pause());
    const order = [0,900,1800,2700].map(time => {
      animations.forEach(animation => { animation.currentTime = time; });
      return frames.filter(frame => getComputedStyle(frame).opacity === '1').map(frame => Number(frame.dataset.frame) + 1);
    });
    animations.forEach(animation => { animation.currentTime = 0; animation.play(); });
    return order;
  });
  check(JSON.stringify(heroOrder) === JSON.stringify([[1],[3],[4],[1]]), 'Hero loops 1 → 3 → 4 → 1 with exactly one visible pose');
  check(await crt.locator('[data-frame]').evaluateAll(frames => frames.every((frame,index) => frame.getAttribute('src').includes(`hero-idle-0${[1,3,4][index]}`))), 'Hero file names follow the pose numbers');
  check(await crt.locator('[data-menu-player]').evaluate(el => {
    const menu = el.parentElement.querySelector('nav').getBoundingClientRect();
    const hero = el.getBoundingClientRect();
    return hero.left > menu.right;
  }), 'Hero stands clear of the menu');
  check(await crt.locator('[data-screen="main"] i').evaluateAll(stars => stars.length === 4 && stars.every(star => Number.parseFloat(getComputedStyle(star).animationDuration) >= 8)), 'Only a few stars pulse slowly');
  check(await page.locator('[data-hardware="active-cartridge"]').count() === 1, 'Exactly one fixed cartridge');
  check(await page.locator('[data-cartridge-action], [data-cartridge-id], [data-readable-toggle], [data-hardware="inactive-cartridge"]').count() === 0, 'Obsolete interactions absent');
  check(await page.locator('[data-hardware="active-cartridge"] button').count() === 0, 'Cartridge is decorative');
  check(await page.locator('[data-controller-input]').count() === 11, 'All controller controls exist');
  check(await page.locator('[data-controller-input], [data-console-control]').evaluateAll(els => els.every(el => el.textContent.trim() === '' && el.getAttribute('aria-label'))), 'Transparent accessible hardware hitboxes');
  check(await page.locator('[data-hardware-art]').count() === 1, 'One authoritative hardware composite');
  check(await page.locator('[data-hardware-art]').evaluate(el => !el.draggable && getComputedStyle(el).userSelect === 'none' && getComputedStyle(el).webkitUserDrag === 'none'), 'Composite cannot select or drag');
  check(await crt.locator('[aria-label="CRT viewport"]').evaluate(el => getComputedStyle(el, '::after').backgroundImage.includes('repeating-linear-gradient')), 'Scanlines retained');
  // Audit every hitbox in controller-local coordinates against measured PNG cap centers.
  const coordinates = { up:[159,860],down:[148.5,915],left:[128,889],right:[188,896],a:[358,955.5],b:[410,943.5],c:[466,934.5],x:[352,903.5],y:[400,889.5],z:[449,884.5],start:[277,897.5] };
  for (const [id,[x,y]] of Object.entries(coordinates)) {
    const delta = await page.locator(`[data-controller-input="${id}"]`).evaluate((el, point) => Math.hypot(el.offsetLeft + el.offsetWidth/2-point[0],el.offsetTop+el.offsetHeight/2-point[1]),[x,y]);
    check(delta < 2, `Hitbox alignment: ${id}`);
  }
  await page.screenshot({ path: 'output/playwright/comet-desktop-en.png' });
  // Temporary inspection outlines are browser-only and never shipped.
  await page.addStyleTag({ content: '[data-controller-input], [data-console-control] { outline: 1px solid #ffcb70 !important; background: #ffc87022 !important; }' });
  await page.screenshot({ path: 'output/playwright/hitbox-audit.png' });
  await page.reload(); await wait('on');
  await page.keyboard.press('ArrowDown'); await page.keyboard.press('Enter');
  check(await rom.getAttribute('data-screen') === 'projects', 'Keyboard opens Projects');
  const labels = ['TRANSPORT CONTROL','FINANCIAL PLATFORM','FACADE BUILDER','REAL-TIME PLATFORM'];
  for (let i=0;i<4;i++) { await choices().nth(i).click(); check(await crt.locator('h2').textContent() === labels[i], 'Project detail'); await back(); }
  await back(); await choices().nth(0).click(); check(await crt.locator('h2').textContent() === 'ABOUT', 'About'); await back();
  await choices().nth(2).click();
  for(let i=0;i<5;i++) { await choices().nth(i).click(); check(await crt.locator('[aria-label="Technologies"] li').count() > 0, 'Toolkit'); }
  await back(); await choices().nth(3).click(); check(await crt.locator('a').count() === 2, 'Contact'); await back();
  await page.locator('[data-controller-input="start"]').click();
  await choices().nth(1).click(); check(await crt.locator('h2').textContent() === 'CONTROLS', 'Options controls'); await back();
  await choices().nth(0).click(); await choices().nth(1).click();
  check(await page.locator('html').getAttribute('lang') === 'ru', 'RU');
  await back(); await back(); await main();
  await page.screenshot({ path: 'output/playwright/comet-desktop-ru.png' });
  for(const i of [0,1,2,3]) { await choices().nth(i).click(); await page.screenshot({ path: `output/playwright/comet-ru-screen-${i}.png` }); await back(); }
  await page.keyboard.press('o'); await choices().nth(0).click(); await choices().nth(0).click(); await back(); await back();
  await choices().nth(0).focus();
  for(const [control,screen] of [['a','about'],['x','about'],['z','options']]) {
    await page.locator(`[data-controller-input="${control}"]`).click(); check(await rom.getAttribute('data-screen') === screen, `Controller ${control}`);
    await page.locator('[data-controller-input="b"]').click();
  }
  await page.locator('[data-controller-input="down"]').click(); await page.locator('[data-controller-input="a"]').click();
  check(await rom.getAttribute('data-screen') === 'projects', 'Controller navigation'); await page.locator('[data-controller-input="y"]').click();
  // Observe the entire reset, not just its final state.
  await page.evaluate(() => { window.__powerStates=[]; const el=document.querySelector('[data-power-state]'); window.__observer=new MutationObserver(() => window.__powerStates.push(el.dataset.powerState)); window.__observer.observe(el,{attributes:true,attributeFilter:['data-power-state']}); });
  await reset.focus(); await page.keyboard.press('Enter');
  await wait('resetting'); check(!(await lineAnimation()).match(/wake|shutdown|collapse/i), 'Reset must not use a CRT line'); check(!(await animation()).match(/wake|collapse/i), 'Reset must not collapse/wake');
  await wait('on'); check(await rom.getAttribute('data-screen') === 'boot', 'ROM boot while hardware on');
  check(!(await lineAnimation()).match(/wake|shutdown/i), 'No wake after reset'); await main();
  const states = await page.evaluate(() => { window.__observer.disconnect(); return window.__powerStates; });
  check(!states.includes('booting') && !states.includes('shuttingDown'), 'Reset lifecycle remains powered');
  check(await reset.evaluate(el => document.activeElement === el), 'Reset preserves focus');
  await power.click(); await wait('shuttingDown'); check((await animation()).includes('collapse'), 'Power collapse retained'); check((await lineAnimation()).includes('shutdownLine'), 'Shutdown line retained');
  await wait('off'); await power.click(); await wait('booting'); check((await lineAnimation()).includes('wakeLine'), 'Power wake retained'); await wait('on');
  await page.emulateMedia({ reducedMotion: 'reduce' });
  check(await page.locator('[data-frame]').evaluateAll(frames => frames.every(el => getComputedStyle(el).animationName === 'none') && getComputedStyle(frames[0]).opacity === '1' && frames.slice(1).every(el => getComputedStyle(el).opacity === '0')), 'Reduced motion uses one neutral hero frame');
  await power.click(); check(await animation() === 'none', 'Reduced motion collapse disabled'); await wait('off'); await power.click(); check(await lineAnimation() === 'none', 'Reduced motion wake disabled'); await wait('on');
  for(const [width,height] of [[390,844],[767,700]]) {
    await page.setViewportSize({width,height}); await page.locator('[data-hardware]').first().waitFor({state:'detached'});
    check(await page.locator('details, nav, button, a').count() === 0, 'Minimal desktop-only fallback');
    check(await page.locator('main').textContent().then(t => t.includes('mouse or trackpad')), 'Desktop input message');
    check(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth), 'Fallback no horizontal overflow');
  }
  await page.setViewportSize({width:1600,height:900}); await main(); await page.emulateMedia({ reducedMotion: 'no-preference' });
  check(await page.evaluate(() => document.documentElement.scrollHeight === innerHeight && document.documentElement.scrollWidth === innerWidth), 'Desktop no overflow');
  check(await rom.evaluate(el => [...el.querySelectorAll('button')].every(b => b.getBoundingClientRect().bottom <= el.getBoundingClientRect().bottom)), 'Menu fits CRT');
  // Inspect the supplied composition at the required desktop sizes after React has resized.
  for(const [width,height] of [[1920,1200],[1920,1080],[1600,900],[1366,768],[1024,768],[900,700],[1000,1000],[1080,1920],[1600,500],[1366,550]]) {
    await page.setViewportSize({width,height});
    await page.waitForFunction(() => Math.abs(document.querySelector('[data-power-state]').getBoundingClientRect().height - Math.min(innerHeight, innerWidth / 1600 * 900)) < 1);
    await page.screenshot({path:`output/playwright/comet-${width}x${height}.png`});
    check(await page.locator('[data-hardware-art]').evaluate(el => {const r = el.getBoundingClientRect(); return r.left >= 0 && r.right <= innerWidth && r.top >= 0 && r.bottom <= innerHeight;}), 'Whole hardware fits desktop');
    check(await crt.locator('[data-frame]').count() === 3, 'Three approved hero poses');
    check(await page.locator('[data-controller-hints]').evaluate(el => Number.parseFloat(getComputedStyle(el).fontSize) >= 16), 'Hints remain readable without scene scaling');
    const hintBounds = await page.locator('[data-controller-hints]').boundingBox();
    check(await page.locator('[data-controller-input], [data-console-control]').evaluateAll((buttons, hint) => buttons.every(button => {
      const r = button.getBoundingClientRect();
      return r.right <= hint.x || r.left >= hint.x + hint.width || r.bottom <= hint.y || r.top >= hint.y + hint.height;
    }), hintBounds), 'Hint card never covers hardware controls');
  }
  const tablet = await page.context().browser().newContext({hasTouch:true,isMobile:true,viewport:{width:1024,height:768}});
  try {
    const touchPage = await tablet.newPage();
    await touchPage.goto(page.url());
    for (const [width,height] of [[768,1024],[1024,768],[1366,1024]]) {
      await touchPage.setViewportSize({width,height});
      await touchPage.locator('h1').waitFor();
      check(await touchPage.evaluate(() => matchMedia('(any-pointer: coarse)').matches && !matchMedia('(any-pointer: fine)').matches), 'Real touch-only input emulation');
      check(await touchPage.locator('[data-hardware]').count() === 0, 'Tablet fallback in both orientations');
    }
  } finally { await tablet.close(); }
  await page.setViewportSize({width:1920,height:1200});
  await page.route('**/*crt-main-background*', route => route.request().resourceType() === 'image' ? route.abort() : route.continue()); await page.reload(); await wait('on'); await choices().nth(0).click(); check(await rom.getAttribute('data-screen') === 'about', 'Missing art does not block navigation'); await page.unroute('**/*crt-main-background*');
  check(issues.length === 0, `Runtime errors: ${issues.join(', ')}`);
  return { heroOrder, projectDetails:4, toolkitCategories:5, controllerHitboxes:11, resetStates:states, runtimeErrors:issues, passed:['single cartridge','no swapping/readable view','keyboard/mouse/controller','EN/RU','powered ROM reset without line','power line/collapse','scanlines','reduced motion','minimal mobile fallback','no overflow','asset failure','readable hints clear of controls','three-pose hero clear of menu','slow star pulses','no decorative CRT HUD'] };
}
