async (page) => {
  const check = (condition, message) => { if (!condition) throw new Error(message); };
  const sizes = () => page.locator('[data-theme] nav button').evaluateAll(buttons => buttons.map(button => {
    const rect = button.getBoundingClientRect();
    const text = [...button.childNodes].find(node => node.nodeType === Node.TEXT_NODE && node.textContent.trim());
    const range = document.createRange();
    range.selectNodeContents(text);
    return {width:rect.width,height:rect.height,font:getComputedStyle(button).fontSize,labelX:range.getBoundingClientRect().x};
  }));
  const equalRows = rows => rows.every(row => Math.abs(row.width-rows[0].width) < .1 && Math.abs(row.height-rows[0].height) < .1 && row.font === rows[0].font && Math.abs(row.labelX-rows[0].labelX) < .1);
  const results = [];
  for (const locale of ['ru','en']) {
    await page.setViewportSize({width:1600,height:900});
    await page.evaluate(locale => localStorage.setItem('portfolio-locale',locale),locale);
    await page.reload();
    await page.locator('[data-power-state="on"]').waitFor();
    await page.keyboard.press('o');
    const optionsBefore = await sizes();
    check(equalRows(optionsBefore), 'Options rows share maximum width, height, font, and text alignment');
    await page.keyboard.press('ArrowDown');
    check(JSON.stringify(optionsBefore) === JSON.stringify(await sizes()), 'Options cursor does not move labels or resize rows');
    await page.screenshot({path:`output/playwright/options-aligned-${locale}.png`});
    await page.keyboard.press('Enter');
    check(await page.locator('[data-theme] h2').evaluate(el => document.activeElement === el && getComputedStyle(el).outlineStyle === 'none'), 'Controls heading is announced without a focus rectangle');
    await page.screenshot({path:`output/playwright/controls-heading-${locale}.png`});
    await page.keyboard.press('Tab');
    check(await page.locator('[data-rom-back]').evaluate(el => document.activeElement === el && getComputedStyle(el).outlineStyle !== 'none'), 'Back retains its visible keyboard focus');
    await page.keyboard.press('Escape');
    await page.keyboard.press('ArrowUp');
    await page.keyboard.press('Enter');
    const languageBefore = await sizes();
    check(equalRows(languageBefore), 'Language rows share maximum width, height, font, and text alignment');
    await page.keyboard.press('ArrowDown');
    check(JSON.stringify(languageBefore) === JSON.stringify(await sizes()), 'Language cursor does not move labels or resize rows');
    await page.screenshot({path:`output/playwright/languages-aligned-${locale}.png`});
    const language = page.locator('[data-theme] nav button');
    await language.nth(0).click();
    check(equalRows(await sizes()), 'Language rows remain equal after activating English');
    await language.nth(1).click();
    check(equalRows(await sizes()), 'Language rows remain equal after activating Russian');
    await page.keyboard.press('Escape');
    await page.keyboard.press('Escape');
    const menu = page.locator('[data-screen="main"] nav button');
    check(await menu.evaluateAll(buttons => buttons.every(button => {
      const rect = button.getBoundingClientRect();
      const text = [...button.childNodes].find(node => node.nodeType === Node.TEXT_NODE && node.textContent.trim());
      const range = document.createRange();range.selectNodeContents(text);
      return range.getBoundingClientRect().right < rect.right - 4;
    })), 'Longest main label stays clear of the selection edge');
    results.push({locale,options:optionsBefore,languages:languageBefore});
  }
  return results;
}
